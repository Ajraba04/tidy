```swift
import Foundation

public struct CleanupCommandOutput: Sendable {
    public let succeeded: Bool
    public let standardOutput: String
    public let standardError: String
    public init(succeeded: Bool, standardOutput: String = "", standardError: String = "") {
        self.succeeded = succeeded
        self.standardOutput = standardOutput
        self.standardError = standardError
    }
}

public protocol CleanupCommandRunning: Sendable {
    func run(executableName: String, arguments: [String], timeout: TimeInterval) async -> CleanupCommandOutput
}

public struct SystemCleanupCommandRunner: CleanupCommandRunning {
    public init() {}
    public func run(executableName: String, arguments: [String], timeout: TimeInterval) async -> CleanupCommandOutput {
        guard let path = ToolLocator.locate(executableName) else {
            return CleanupCommandOutput(succeeded: false, standardError: "\(executableName) not found")
        }
        do {
            let output = try await ProcessRunner.run(executableURL: URL(fileURLWithPath: path), arguments: arguments, timeout: timeout)
            return CleanupCommandOutput(succeeded: output.succeeded, standardOutput: output.standardOutput, standardError: output.standardError)
        } catch {
            return CleanupCommandOutput(succeeded: false, standardError: error.localizedDescription)
        }
    }
}
```

Transport is an effect-only dependency; a supplied recorder can return success but cannot change denial. It is not a new authorized mutation surface and must not be exposed to catalog data as a callable closure. The underlying old ProcessRunner timeout semantics remain unsuitable for accepted mutations (T04/S12/S16); because no call reaches this transport after quarantine, this task does not claim to fix them.

3. For `.unavailable`, emit unavailable with its reason, no locate/discover.
